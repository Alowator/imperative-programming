// sum.c
long long pref[100000 + 1];
int size;

void Init(const int *arr, int n) {
    size = n;
    pref[0] = 0;
    for (int i = 1; i <= n; i++) {
        pref[i] = pref[i - 1] + arr[i - 1];
    }
}

long long Sum(int l, int r) {
    long long res = pref[r] - pref[l];
    return res;
}

// query.c
extern int size;

long long Sum(int l, int r);

int Query(int l, long long sum) {
    int left = l;
    int r = size;
    int x;
    while ( l < r ) {
        x = (l + r + 1) / 2;
        long long s = Sum(left, x);
        if ( Sum(left, x) <= sum ) {
            l = x;
        } else {
            r = x - 1;
        }
    }
    return l;
}
