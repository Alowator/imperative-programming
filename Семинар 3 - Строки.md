1. Найти подстроку в строке, без учета регистра

```c
char* findSubstringIgnoreCase(const char* str, const char* substr) {
    for (const char* p = str; *p != '\0'; ++p) {
        const char* s1 = p;
        const char* s2 = substr;

        while (*s1 && *s2 && tolower((unsigned char)*s1) == tolower((unsigned char)*s2)) {
            ++s1;
            ++s2;
        }

        if (*s2 == '\0') {
            return (char*)p;
        }
    }

    return 0;
}
```
